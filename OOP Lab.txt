import java.util.Scanner;

public class Main {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        // Canteen Menu
        String[] menu = {
            "Burger",
            "Pizza",
            "Fried Chicken",
            "Spaghetti",
            "French Fries"
        };

        double[] price = {
            100.00,
            150.00,
            120.00,
            90.00,
            60.00
        };

        int totalQuantity = 0;
        double totalAmount = 0.00;
        boolean isStudent = false;

        System.out.println("==================================");
        System.out.println("        SCHOOL CANTEEN MENU       ");
        System.out.println("==================================");

        for (int i = 0; i < menu.length; i++) {
            System.out.printf("%d. %-20s $%.2f%n",
                    i + 1, menu[i], price[i]);
        }

        System.out.println("==================================");

        char orderAgain = 'Y';

        while (orderAgain == 'Y') {

            System.out.print("Enter item number (1-5): ");
            int itemNumber = input.nextInt();

            System.out.print("Enter quantity (1-10): ");
            int quantity = input.nextInt();

            System.out.print("Are you a student? (Y/N): ");
            char student = input.next().toUpperCase().charAt(0);

            // Check if the order is valid
            if (itemNumber < 1 || itemNumber > 5) {

                System.out.println("Invalid item number!");
                System.out.println("Order was not added.");

            } else if (quantity < 1 || quantity > 10) {

                System.out.println("Invalid quantity!");
                System.out.println("Order was not added.");

            } else if (student != 'Y' && student != 'N') {

                System.out.println("Invalid student status!");
                System.out.println("Order was not added.");

            } else {

                // Valid order
                isStudent = (student == 'Y');

                double orderAmount =
                        price[itemNumber - 1] * quantity;

                totalQuantity += quantity;
                totalAmount += orderAmount;

                System.out.printf(
                        "%s x %d = $%.2f%n",
                        menu[itemNumber - 1],
                        quantity,
                        orderAmount
                );

                System.out.println("Order added successfully!");
            }

            // Ask for another order
            System.out.print("Do you want to order again? (Y/N): ");
            orderAgain = input.next().toUpperCase().charAt(0);

            while (orderAgain != 'Y' && orderAgain != 'N') {
                System.out.println("Invalid choice!");
                System.out.print("Please enter Y or N: ");
                orderAgain = input.next().toUpperCase().charAt(0);
            }

            System.out.println();
        }

        // Determine deduction
        double deductionRate;

        if (isStudent && totalAmount >= 500) {
            deductionRate = 0.15;
        } else if (isStudent) {
            deductionRate = 0.10;
        } else if (totalAmount >= 500) {
            deductionRate = 0.05;
        } else {
            deductionRate = 0.00;
        }

        // Final calculations
        double totalDeduction = totalAmount * deductionRate;
        double finalAmount = totalAmount - totalDeduction;

        // Final receipt
        System.out.println("==================================");
        System.out.println("           FINAL RECEIPT          ");
        System.out.println("==================================");

        System.out.println("Total quantity of items: "
                + totalQuantity);

        System.out.printf(
                "Total amount before deduction: $%.2f%n",
                totalAmount
        );

        System.out.printf(
                "Total deduction: $%.2f%n",
                totalDeduction
        );

        System.out.printf(
                "Final amount to pay: $%.2f%n",
                finalAmount
        );

        System.out.println("==================================");
        System.out.println("       Thank you for ordering!    ");
        System.out.println("==================================");

        input.close();
    }
}
